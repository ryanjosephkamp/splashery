const sources = Object.fromEntries(["before", "after"].map((mode) =>
  [mode, fs.readFileSync(new URL(`./ui-${mode}.js`, import.meta.url), "utf8")]));
const test = base.extend({
  uiMode: ["before", { option: true }],
  page: async ({ page, uiMode }, use) => {
    await page.route("**/src/ui.js", (route) => route.fulfill({
      status: 200, contentType: "text/javascript", body: sources[uiMode],
    }));
    await page.addInitScript(() => {
      const rows = (window.__shrFocus = []);
      const snapshot = (kind, dialog) => {
        const active = document.activeElement;
        const pill = document.getElementById("rec-pill");
        rows.push({ kind, at: performance.now(), dialog: dialog?.id,
          open: dialog?.open, active: active?.id || active?.tagName,
          activeInPill: !!pill?.contains(active), pillHidden: pill?.hidden });
      };
      for (const method of ["showModal", "close"]) {
        const native = HTMLDialogElement.prototype[method];
        HTMLDialogElement.prototype[method] = function (...args) {
          snapshot(method + ":before", this);
          const result = native.apply(this, args);
          snapshot(method + ":after", this);
          return result;
        };
      }
      for (const type of ["focusin", "focusout", "close", "cancel"])
        document.addEventListener(type, (event) => snapshot(type, document.getElementById("rec-preview")), true);
    });
    await use(page);
  },
});

async function open(page, width, height) {
  await page.setViewportSize({ width, height });
  await page.goto("/?renderer=webgl2&profile=weak&labs=0");
  await page.waitForSelector("body[data-ready='true']", { timeout: 180_000 });
}

async function expectRecordFocused(page) {
  const got = await page.evaluate(() => {
    const r = document.getElementById("record-start");
    return { shown: !!r.offsetParent && r.getBoundingClientRect().height > 0, focused: document.activeElement === r, body: document.activeElement === document.body }; // prettier-ignore
  });
  if (got.shown) expect(got.focused).toBe(true);
  else expect(got.body || got.focused).toBe(true);
}
